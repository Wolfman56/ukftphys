import numpy as np
import matplotlib.pyplot as plt
from scipy.linalg import expm

# Parameters
N = 201                  # lattice sites (odd for center)
t_hop = 1.0              # hopping strength
dt = 0.2                 # discrete time step per tick
T_ticks = 200            # number of choice ticks
M_particles = 500        # ensemble size for statistics

# Tight-binding Hamiltonian (tridiagonal + periodic)
diag = np.zeros(N)
off_diag = -t_hop * np.ones(N)
H = np.diag(diag) + np.diag(off_diag[:-1], 1) + np.diag(off_diag[:-1], -1)
H[0, -1] = H[-1, 0] = -t_hop  # periodic

# Double-slit initial wavefunction: left Gaussian split by phase
x = np.arange(N)
center_left = N // 4
width = N / 20
psi0 = np.exp(-((x - center_left)**2) / (2 * width**2))
psi0 /= np.linalg.norm(psi0)  # normalize

# Mimic two slits: phase flip in lower half of packet + narrow barrier
slit_mask = np.exp(1j * np.pi * (x > center_left + width / 2))  # crude phase for interference
psi0 *= slit_mask
psi0 /= np.linalg.norm(psi0)

# Precompute unitary propagator
U = expm(-1j * H * dt)

# Function to build choice matrix C(psi)
def build_choice_matrix(psi):
    rho = np.abs(psi)**2 + 1e-12  # avoid divide-by-zero
    # Current J_k from k to k+1
    J = 2 * np.imag(np.conj(psi) * np.roll(psi, -1))  # positive = right flow
    # Adaptive beta to keep probs safe
    max_flow = np.max(np.abs(J) / rho)
    beta = 0.8 / (max_flow * dt + 1e-8) if max_flow > 0 else 1.0
    
    p_right = beta * dt * np.maximum(J, 0) / rho
    p_left = beta * dt * np.maximum(-np.roll(J, 1), 0) / rho  # incoming from left → left hop probability
    p_stay = 1 - p_right - p_left
    p_stay = np.maximum(p_stay, 0)  # renormalize if needed
    norm = p_stay + p_right + p_left
    p_stay /= norm
    p_right /= norm
    p_left /= norm
    
    # Build tridiagonal stochastic matrix (rows = to, columns = from)
    C = np.zeros((N, N))
    np.fill_diagonal(C, p_stay)
    np.fill_diagonal(C[1:, :], p_right[:-1])   # superdiagonal
    np.fill_diagonal(C[:, 1:], p_left[:-1])    # subdiagonal
    # Periodic boundaries
    C[0, -1] = p_left[-1]
    C[-1, 0] = p_right[0]
    return C

# Simulate ensemble of ontological particles
psi = psi0.copy()
positions_over_time = np.zeros((M_particles, T_ticks + 1), dtype=int)

# Initial positions sampled from |psi0|^2
initial_probs = np.abs(psi0)**2
initial_probs /= initial_probs.sum()
positions_over_time[:, 0] = np.random.choice(N, size=M_particles, p=initial_probs)

for tick in range(1, T_ticks + 1):
    # Unitary pilot update
    psi = U @ psi
    psi /= np.linalg.norm(psi)  # numerical stability
    
    # Build choice matrix
    C = build_choice_matrix(psi)
    
    # Update each particle stochastically
    for p in range(M_particles):
        current_site = positions_over_time[p, tick-1]
        next_probs = C[:, current_site]  # column = transition from current_site
        next_site = np.random.choice(N, p=next_probs)
        positions_over_time[p, tick] = next_site

# Plot results
fig, ax = plt.subplots(2, 1, figsize=(10, 8))

# Theoretical |psi|^2 at final time (continuum reference)
psi_final = psi.copy()
ax[0].plot(np.abs(psi_final)**2, label='Pilot |ψ|² (theory)', lw=2)
ax[0].hist(positions_over_time[:, -1], bins=N, density=True, alpha=0.6, label='Empirical density (choices)')
ax[0].set_title('Final Density: Theory vs Discrete Choices')
ax[0].legend()

# Sample trajectories
for p in range(min(20, M_particles)):
    ax[1].plot(positions_over_time[p, :], alpha=0.5)
ax[1].set_title('Sample Ontological Trajectories (Discrete Guided Choices)')
ax[1].set_xlabel('Choice Tick')
ax[1].set_ylabel('Lattice Site')

plt.tight_layout()
plt.show()

def build_choice_matrix(psi, alpha=0.0):
    rho = np.abs(psi)**2 + 1e-12
    J = 2 * np.imag(np.conj(psi) * np.roll(psi, -1))
    max_flow = np.max(np.abs(J) / rho)
    beta = 0.8 / (max_flow * dt + 1e-8) if max_flow > 0 else 1.0
    
    # Base guidance
    p_right = beta * dt * np.maximum(J, 0) / rho
    p_left = beta * dt * np.maximum(-np.roll(J, 1), 0) / rho
    p_stay = 1.0 - p_right - p_left
    p_stay = np.maximum(p_stay, 0.0)
    
    # Entropic bias
    if alpha > 0:
        rho_right = np.roll(rho, -1)
        rho_left = np.roll(rho, 1)
        bias_right = alpha * np.maximum(rho_right - rho, 0)
        bias_left = alpha * np.maximum(rho_left - rho, 0)
        p_right += bias_right
        p_left += bias_left
    
    # Always renormalize for safety
    total = p_right + p_left + p_stay + 1e-15
    p_right /= total
    p_left /= total
    p_stay /= total
    
    return p_left, p_stay, p_right

import numpy as np
import matplotlib.pyplot as plt
from scipy.linalg import expm

np.random.seed(42)  # repro

# Parameters
N = 201
t_hop = 1.0
dt = 0.2
T_ticks = 200
M_particles = 1000
alpha_entropic = 0.2  # tune for gravity strength
use_entropic = True

# Hamiltonian + propagator
diag = np.zeros(N)
off_diag = -t_hop * np.ones(N)
H = np.diag(diag) + np.diag(off_diag[:-1], 1) + np.diag(off_diag[:-1], -1)
H[0, -1] = H[-1, 0] = -t_hop
U = expm(-1j * H * dt)

# Initial ψ
x = np.arange(N)
center_left = N // 4
width = N / 20.0
gaussian = np.exp(-((x - center_left)**2) / (2 * width**2))
gaussian /= np.linalg.norm(gaussian)
k0 = 0.8
momentum_boost = np.exp(1j * k0 * x)
slit_mask = np.exp(1j * np.pi * (x > center_left + width / 2))
psi0 = gaussian * slit_mask * momentum_boost
psi0 /= np.linalg.norm(psi0)

def build_choice_matrix(psi, alpha=0.0):
    rho = np.abs(psi)**2 + 1e-12
    J = 2 * np.imag(np.conj(psi) * np.roll(psi, -1))
    max_flow = np.max(np.abs(J) / rho)
    beta = 0.8 / (max_flow * dt + 1e-8) if max_flow > 0 else 1.0
    
    p_right = beta * dt * np.maximum(J, 0) / rho
    p_left = beta * dt * np.maximum(-np.roll(J, 1), 0) / rho
    p_stay = 1.0 - p_right - p_left
    p_stay = np.maximum(p_stay, 0.0)
    
    if alpha > 0:
        rho_right = np.roll(rho, -1)
        rho_left = np.roll(rho, 1)
        bias_right = alpha * np.maximum(rho_right - rho, 0)
        bias_left = alpha * np.maximum(rho_left - rho, 0)
        p_right += bias_right
        p_left += bias_left
    
    total = p_right + p_left + p_stay + 1e-15
    p_right /= total
    p_left /= total
    p_stay /= total
    
    return p_left, p_stay, p_right

# Initial particles
pk = np.abs(psi0)**2
pk /= pk.sum()
positions = np.random.choice(np.arange(N), size=M_particles, p=pk)

for t in range(T_ticks):
